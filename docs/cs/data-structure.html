第一章 线性表

1.1 线性表的定义和基本操作

线性表的定义

具有相同特性的数据元素的有限序列。

基本操作

  • InitList(&L)：初始化
  • DestroyList(&L)：销毁
  • ClearList(&L)：清空
  • ListEmpty(L)：判空
  • ListLength(L)：求长度
  • GetElem(L, i, &e)：获取第i个元素
  • LocateElem(L, e, compare())：查找元素位置
  • ListInsert(&L, i, e)：插入元素
  • ListDelete(&L, i, &e)：删除元素
  • ListTraverse(L, visit())：遍历

1.2 线性表的顺序表示和实现

顺序表的定义

用一组地址连续的存储单元依次存储线性表的数据元素。

顺序表的实现

// 顺序表结构定义
typedef struct {
    int data[MAXSIZE];
    int length;
} SqList;

// 插入操作
int ListInsert(SqList *L, int i, int e) {
    if (i < 1 || i > L->length + 1) return 0;
    if (L->length >= MAXSIZE) return 0;
    for (int j = L->length; j >= i; j--) {
        L->data[j] = L->data[j-1];
    }
    L->data[i-1] = e;
    L->length++;
    return 1;
}

顺序表的优缺点

优点 缺点
随机访问，时间复杂度O(1) 插入删除效率低，时间复杂度O(n)
存储密度高 需要预先分配空间，可能浪费

1.3 线性表的链式表示和实现

单链表

// 单链表节点结构
typedef struct LNode {
    int data;
    struct LNode *next;
} LNode, *LinkList;

// 头插法建立链表
void CreateListHead(LinkList *L, int n) {
    *L = (LinkList)malloc(sizeof(LNode));
    (*L)->next = NULL;
    for (int i = n; i > 0; i--) {
        LNode *p = (LNode*)malloc(sizeof(LNode));
        p->data = i;
        p->next = (*L)->next;
        (*L)->next = p;
    }
}

双链表

// 双链表节点结构
typedef struct DNode {
    int data;
    struct DNode *prior, *next;
} DNode, *DLinkList;

循环链表

单循环链表：尾节点指针指向头节点；双循环链表：头节点前驱指向尾节点，尾节点后继指向头节点。

第二章 栈和队列

2.1 栈

栈的定义和特点

后进先出（LIFO），只能在一端进行插入和删除操作。

栈的顺序存储结构

// 顺序栈结构
typedef struct {
    int data[MAXSIZE];
    int top;
} SqStack;

// 入栈
int Push(SqStack *S, int e) {
    if (S->top == MAXSIZE - 1) return 0;
    S->data[++S->top] = e;
    return 1;
}

// 出栈
int Pop(SqStack *S, int *e) {
    if (S->top == -1) return 0;
    *e = S->data[S->top--];
    return 1;
}

栈的应用

  • 表达式求值（中缀转后缀）
  • 括号匹配
  • 函数调用
  • 递归实现

2.2 队列

队列的定义和特点

先进先出（FIFO），一端插入（队尾），一端删除（队头）。

循环队列

// 循环队列结构
typedef struct {
    int data[MAXSIZE];
    int front, rear;
} SqQueue;

// 入队
int EnQueue(SqQueue *Q, int e) {
    if ((Q->rear + 1) % MAXSIZE == Q->front) return 0;
    Q->data[Q->rear] = e;
    Q->rear = (Q->rear + 1) % MAXSIZE;
    return 1;
}

// 出队
int DeQueue(SqQueue *Q, int *e) {
    if (Q->front == Q->rear) return 0;
    *e = Q->data[Q->front];
    Q->front = (Q->front + 1) % MAXSIZE;
    return 1;
}

队列的应用

  • 广度优先搜索
  • 任务调度
  • 缓冲区管理

注意：循环队列中，判空条件是 front == rear，判满条件是 (rear + 1) % MAXSIZE == front。

第三章 树和二叉树

3.1 树的基本概念

树的定义

n个节点的有限集合，n=0时为空树；有且仅有一个根节点；其余节点分为m个互不相交的子树。

基本术语

  • 节点的度：节点拥有的子树数
  • 树的度：树中节点的最大度数
  • 叶子节点：度为0的节点
  • 分支节点：度不为0的节点
  • 节点的层次：根节点为第1层
  • 树的深度：树中节点的最大层次

3.2 二叉树

二叉树的定义

每个节点最多有两个子树，左子树和右子树有顺序。

特殊二叉树

  • 满二叉树：所有节点都有两个孩子，叶子在同一层
  • 完全二叉树：除最后一层外，每层都满，最后一层节点从左到右排列
  • 二叉搜索树：左子树所有节点值小于根，右子树所有节点值大于根
  • 平衡二叉树：左右子树高度差不超过1

二叉树的性质

  • 第i层最多有2^(i-1)个节点
  • 深度为k的二叉树最多有2^k - 1个节点
  • n₀ = n₂ + 1（叶子节点数=度为2的节点数+1）

二叉树的存储结构

// 二叉链表结构
typedef struct BiTNode {
    int data;
    struct BiTNode *lchild, *rchild;
} BiTNode, *BiTree;

3.3 二叉树的遍历

前序遍历

void PreOrderTraverse(BiTree T) {
    if (T) {
        printf("%d ", T->data);
        PreOrderTraverse(T->lchild);
        PreOrderTraverse(T->rchild);
    }
}

中序遍历

void InOrderTraverse(BiTree T) {
    if (T) {
        InOrderTraverse(T->lchild);
        printf("%d ", T->data);
        InOrderTraverse(T->rchild);
    }
}

后序遍历

void PostOrderTraverse(BiTree T) {
    if (T) {
        PostOrderTraverse(T->lchild);
        PostOrderTraverse(T->rchild);
        printf("%d ", T->data);
    }
}

层序遍历

void LevelOrderTraverse(BiTree T) {
    SqQueue Q;
    InitQueue(&Q);
    if (T) EnQueue(&Q, T);
    while (!QueueEmpty(Q)) {
        DeQueue(&Q, &p);
        printf("%d ", p->data);
        if (p->lchild) EnQueue(&Q, p->lchild);
        if (p->rchild) EnQueue(&Q, p->rchild);
    }
}

3.4 二叉搜索树

查找操作

BiTree SearchBST(BiTree T, int key) {
    if (!T || T->data == key) return T;
    if (key < T->data) return SearchBST(T->lchild, key);
    else return SearchBST(T->rchild, key);
}

插入操作

int InsertBST(BiTree *T, int key) {
    if (!*T) {
        *T = (BiTree)malloc(sizeof(BiTNode));
        (*T)->data = key;
        (*T)->lchild = (*T)->rchild = NULL;
        return 1;
    }
    if (key == (*T)->data) return 0;
    else if (key < (*T)->data) return InsertBST(&((*T)->lchild), key);
    else return InsertBST(&((*T)->rchild), key);
}

4.3 图的遍历

深度优先搜索（DFS）

void DFS(GraphAdjList G, int v) {
    visited[v] = 1;
    printf("%d ", G.adjList[v].data);
    EdgeNode *p = G.adjList[v].firstedge;
    while (p) {
        if (!visited[p->adjvex]) DFS(G, p->adjvex);
        p = p->next;
    }
}

广度优先搜索（BFS）

void BFS(GraphAdjList G, int v) {
    SqQueue Q;
    InitQueue(&Q);
    visited[v] = 1;
    printf("%d ", G.adjList[v].data);
    EnQueue(&Q, v);
    while (!QueueEmpty(Q)) {
        DeQueue(&Q, &v);
        EdgeNode *p = G.adjList[v].firstedge;
        while (p) {
            if (!visited[p->adjvex]) {
                visited[p->adjvex] = 1;
                printf("%d ", G.adjList[p->adjvex].data);
                EnQueue(&Q, p->adjvex);
            }
            p = p->next;
        }
    }
}

第五章 查找

5.1 顺序查找

int SeqSearch(int a[], int n, int key) {
    for (int i = 0; i < n; i++) {
        if (a[i] == key) return i;
    }
    return -1;
}

5.2 折半查找

int BinarySearch(int a[], int n, int key) {
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = (low + high) / 2;
        if (a[mid] == key) return mid;
        else if (a[mid] > key) high = mid - 1;
        else low = mid + 1;
    }
    return -1;
}

第六章 排序

6.1 插入排序

直接插入排序

void InsertSort(int a[], int n) {
    for (int i = 1; i < n; i++) {
        int temp = a[i];
        int j = i - 1;
        while (j >= 0 && a[j] > temp) {
            a[j+1] = a[j];
            j--;
        }
        a[j+1] = temp;
    }
}

6.2 交换排序

冒泡排序

void BubbleSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j+1]) {
                int temp = a[j];
                a[j] = a[j+1];
                a[j+1] = temp;
            }
        }
    }
}

快速排序

void QuickSort(int a[], int low, int high) {
    if (low < high) {
        int pivot = a[low];
        int i = low, j = high;
        while (i < j) {
            while (i < j && a[j] >= pivot) j--;
            a[i] = a[j];
            while (i < j && a[i] <= pivot) i++;
            a[j] = a[i];
        }
        a[i] = pivot;
        QuickSort(a, low, i - 1);
        QuickSort(a, i + 1, high);
    }
}

6.3 选择排序

简单选择排序

void SelectSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[min]) min = j;
        }
        if (min != i) {
            int temp = a[i];
            a[i] = a[min];
            a[min] = temp;
        }
    }
}

算法 平均时间 最坏时间 空间复杂度 稳定性

冒泡排序 O(n²) O(n²) O(1) 稳定
快速排序 O(nlogn) O(n²) O(logn) 不稳定
